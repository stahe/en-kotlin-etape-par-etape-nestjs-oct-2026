// A HTTP client: an object that is expensive to create, which is created UNE times
val client = OkHttpClient()

// the request: a function SUSPENDUE, which returns the CAPTCHA or throws an exception
suspend fun lireCaptcha(): Captcha = withContext(Dispatchers.IO) {
    val requete = Request.Builder().url("${Config.urlServeur}/api/captcha").build()
    client.newCall(requete).execute().use { reponse ->
         // OkHttp does not throw an exception for an error response (4xx,
         // 5xx): the status code is checked
        if (!reponse.isSuccessful) {
            throw Exception("le serveur a répondu ${reponse.code} ${reponse.message}")
        }
        val json = Json.parseToJsonElement(reponse.body!!.string()).jsonObject
        Captcha(json["id"]!!.jsonPrimitive.content, json["svg"]!!.jsonPrimitive.content)
    }
}

@Composable
fun PageCaptcha() {
    var captcha by remember { mutableStateOf<Captcha?>(null) }
    var enCours by remember { mutableStateOf(true) }
    var erreur by remember { mutableStateOf<String?>(null) }
    var nbRequetes by remember { mutableIntStateOf(0) }
     // the DEMANDE number: changing it triggers the effect below
    var demande by remember { mutableIntStateOf(0) }

     // the request, as soon as the screen appears (request = 0), then with each

     // previous one, the previous coroutine is ANNULÉE: its response, now stale,

        enCours = false

        Spacer(Modifier.height(12.dp))

                Text("Impossible d'obtenir un captcha : $it.\nLe serveur JSON RdvMedecins est-il lancé (port 8080) ?", Modifier.padding(12.dp))

        if (c != null && !enCours) {

                hauteur = 60.dp,

            Spacer(Modifier.height(8.dp))
            Text("id : ${c.id}", fontFamily = FontFamily.Monospace, fontSize = 11.sp)
        }
        Spacer(Modifier.height(12.dp))
        Button(onClick = { demande++ }, enabled = !enCours) { Text("Another image") }
        Spacer(Modifier.height(8.dp))
        Text("$nbRequetes requête(s) réussie(s)", color = Color.Gray)

package fr.stahe.exemples.commun

import android.graphics.Bitmap
import android.graphics.Canvas
import androidx.compose.foundation.Image
import androidx.compose.foundation.layout.height
import androidx.compose.runtime.Composable
import androidx.compose.runtime.remember
import androidx.compose.ui.Modifier
import androidx.compose.ui.graphics.asImageBitmap
import androidx.compose.ui.layout.ContentScale
import androidx.compose.ui.unit.Dp
import androidx.compose.ui.unit.dp
import com.caverock.androidsvg.SVG

@Composable
fun ImageSvg(svg: String, modifier: Modifier = Modifier, hauteur: Dp = 50.dp, description: String? = null) {
     // The drawing is redrawn only if the text SVG changes (remember(svg))
    val image = remember(svg) {
        val document = SVG.getFromString(svg)
         // drawn three times larger than its “natural” size: a sharp image
        val echelle = 3f
        val largeur = (document.documentWidth.takeIf { it > 0 } ?: 150f)
        val haut = (document.documentHeight.takeIf { it > 0 } ?: 50f)
        val bitmap = Bitmap.createBitmap((largeur * echelle).toInt(), (haut * echelle).toInt(), Bitmap.Config.ARGB_8888)
        val toile = Canvas(bitmap)
        toile.scale(echelle, echelle)
        document.renderToCanvas(toile)
        bitmap.asImageBitmap()
    }
    Image(image, contentDescription = description, contentScale = ContentScale.Fit, modifier = modifier.height(hauteur))
}

// common/ClientHttp.kt - a client HTTP that handles COOKIES
// -------------------------------------------------------------------------
// The server RdvMedecins stores the token JWT in a COOKIE (httpOnly).
// In a browser, it is the browser that stores cookies and
// sends back with every request: the application code has nothing to do with it.
// On a phone, there is no PAS browser: it’s up to the app to
// fulfill this role, that is:
//    1. read the "Set-Cookie" header in each response and store the cookies;
//    2. send them back in the "Cookie" header of each request.
// The OkHttp library does this for us if we provide it with a CookieJar
// (a “cookie jar”): a class with two functions, saveFromResponse
// (put away) and loadForRequest (take out). This one also stores the cookies
// on the device (SharedPreferences): you remain logged in when you close
// and then restart the app.
// =========================================================================
package fr.stahe.exemples.commun

import android.content.Context
import android.content.SharedPreferences
import androidx.core.content.edit
import okhttp3.Cookie
import okhttp3.CookieJar
import okhttp3.HttpUrl
import okhttp3.HttpUrl.Companion.toHttpUrl
import okhttp3.OkHttpClient
import java.util.concurrent.TimeUnit

class CookiesSurLAppareil(private val stockage: SharedPreferences) : CookieJar {
     // cookie name -> the cookie (its value, domain, expiration date, etc.)
    private val cookies = mutableMapOf<String, Cookie>()

    init {
         // cookies stored from a previous session: each
         // is stored in the form of its Set-Cookie header
        val base = Config.urlServeur.toHttpUrl()
        for (texte in stockage.getStringSet(CLE, emptySet()).orEmpty()) {
            Cookie.parse(base, texte)?.let { cookies[it.name] = it }
        }
    }

     // The response contains cookies: we store them. A cookie that has already expired
     // (logout: the server deletes it this way) is removed
    @Synchronized
    override fun saveFromResponse(url: HttpUrl, cookies: List<Cookie>) {
        for (c in cookies) {
            if (c.expiresAt < System.currentTimeMillis() || c.value.isEmpty()) {
                this.cookies.remove(c.name)
            } else {
                this.cookies[c.name] = c
            }
        }
        stockage.edit { putStringSet(CLE, this@CookiesSurLAppareil.cookies.values.map { it.toString() }.toSet()) }
    }

     // A request is sent: the relevant cookies (address, path), which have not expired
    @Synchronized
    override fun loadForRequest(url: HttpUrl): List<Cookie> =
        cookies.values.filter { it.expiresAt >= System.currentTimeMillis() && it.matches(url) }

    companion object {
        private const val CLE = "rdvmedecins.cookies"
    }
}

// the client HTTP examples: a single one for the entire application
fun creerClientHttp(context: Context): OkHttpClient = OkHttpClient.Builder()
    .cookieJar(CookiesSurLAppareil(context.getSharedPreferences("exemples", Context.MODE_PRIVATE)))
     // No response after 10 seconds: the request fails
    .connectTimeout(10, TimeUnit.SECONDS)
    .readTimeout(10, TimeUnit.SECONDS)
    .build()

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue
import kotlinx.coroutines.CancellationException

class Commande {
    var enCours by mutableStateOf(false)
        private set
    var erreur by mutableStateOf<String?>(null)
        private set

     // performs the action, unless another one is already in progress
    suspend fun executer(action: suspend () -> Unit) {
        if (enCours) return // no double submission (double-tapping the button)
        enCours = true
        erreur = null
        try {
            action()
        } catch (e: CancellationException) {
            throw e // A canceled coroutine must remain canceled
        } catch (e: Exception) {
            erreur = e.message ?: e.toString() // the error is GARDÉE, to be displayed
        } finally {
             // finally: executed in all cases (success or error) in TOUS
            enCours = false
        }
    }

    fun effacerErreur() {
        erreur = null
    }
}

// common/ErreurApi.kt - UNE class for all errors in API
// -------------------------------------------------------------------------
// All error responses from the RdvMedecins server have the same format:
//    { "statusCode": 409, "cle": "ERRORS.LOGIN_TAKEN", "params": {...}, "fields": {...} }
// But a request can also fail (AUTREMENT): no response at all
// (network down, server down, incorrect address), or a response that
// doesn’t come from our server (an error page from a proxy). We convert
// all of this into an exception class, ErreurApi, which contains the key for
// translation key sent by the server (and, if any, its parameters and
// error for each field).
// =========================================================================
package fr.stahe.exemples.commun

import kotlinx.serialization.json.Json
import kotlinx.serialization.json.JsonObject
import kotlinx.serialization.json.JsonPrimitive
import kotlinx.serialization.json.jsonObject

class ErreurApi(
    val statusCode: Int, // 0: the server did not respond
    val cle: String, // e.g., "ERRORS.LOGIN_TAKEN"
    val params: Map<String, String> = emptyMap(),
    val champs: Map<String, String> = emptyMap(),
) : Exception("$statusCode $cle") {

    override fun toString() = "$statusCode $cle"

     // companion object: what belongs to CLASSE (Java "static" variables)
    companion object {
         // an error response (4xx, 5xx) -> a ErreurApi
        fun depuisReponse(statusCode: Int, corps: String): ErreurApi {
             // runCatching: the result, or null if the body is not JSON
             // (in which case, it was not our server that responded)
            val json = runCatching { Json.parseToJsonElement(corps).jsonObject }.getOrNull()
            val cle = (json?.get("cle") as? JsonPrimitive)?.content
            if (json == null || cle == null) {
                return ErreurApi(statusCode, if (statusCode >= 500) "ERRORS.SERVER_UNREACHABLE" else "ERRORS.SERVER_ERROR")
            }
            return ErreurApi(statusCode, cle, chaines(json["params"]), chaines(json["champs"]))
        }

         // a JSON object { "name": "ERRORS..." } -> Map<String, String> (empty otherwise)
        private fun chaines(json: Any?): Map<String, String> =
            (json as? JsonObject)?.mapValues { (_, v) -> (v as? JsonPrimitive)?.content ?: v.toString() } ?: emptyMap()
    }
}

        private set

        private set
    var idClient by mutableStateOf<Int?>(null)
        private set

        private set

        private set

    fun changerUtilisateur(u: Utilisateur?) {
        utilisateur = u
        medecins = emptyList()
        clients = emptyList()
        agenda = null
        idMedecin = null
        cache.clear() // permissions have changed: forget everything
        if (u == null) return

            try {
                medecins = rdvmedecins.getMedecins()
                if (gestionnaire) clients = rdvmedecins.getClients() // a USER would receive a 403
            } catch (e: ErreurApi) {
                message = Message(false, e.toString())
            }
        }
    }

        idMedecin = id
        message = null
        lireAgenda()
    }

    fun choisirJour(d: LocalDate) {
        jour = d.toString()
        message = null
        lireAgenda()
    }

    fun choisirClient(id: Int?) {
        idClient = id
    }

    fun reserver(c: CreneauAgenda) = agir("Créneau ${c.horaire} réservé.") {
        rdvmedecins.reserver(jour, c.id, if (gestionnaire) idClient else null)
    }

     // an action, then TOUJOURS a review of the agenda: SERVEUR is the authoritative version
    private fun agir(succes: String, action: suspend () -> Unit) {
        viewModelScope.launch {
            message = try {
                action()
                Message(true, succes)
            } catch (e: ErreurApi) {
                Message(false, e.toString())
            }
            lireAgenda()
        }
    }

    private fun lireAgenda() {
        val id = idMedecin ?: return
        val cle = "$id/$jour"
         // an older read is still in progress: we cancel it
        lecture?.cancel()
         // first the cache (or nothing), then the read
        agenda = cache[cle]
        chargement = true
        lecture = viewModelScope.launch {
            try {
                val lu = rdvmedecins.getAgenda(id, jour)
                cache[cle] = lu
                agenda = lu
            } catch (e: ErreurApi) {
                agenda = null
                message = Message(false, e.toString())
            }
            chargement = false
        }
    }
}

@Composable
fun BarreConnexion(api: Api, onUtilisateurChange: (Utilisateur?) -> Unit) {
    var utilisateur by remember { mutableStateOf<Utilisateur?>(null) }
    var captcha by remember { mutableStateOf<Captcha?>(null) }
    var login by rememberSaveable { mutableStateOf("admin") }
    var password by rememberSaveable { mutableStateOf("") }
    var reponseCaptcha by rememberSaveable { mutableStateOf("") }
    var erreur by remember { mutableStateOf<String?>(null) }
    val portee = rememberCoroutineScope()

    fun changer(u: Utilisateur?) {
        utilisateur = u
        onUtilisateurChange(u)
    }

    suspend fun nouveauCaptcha() {
        captcha = api.get("/captcha").lire<Captcha>()
        reponseCaptcha = ""
    }

     // at startup: are we already logged in?
    LaunchedEffect(Unit) {
        try {
            changer(api.get("/auth/moi").lire<Utilisateur>())
        } catch (e: ErreurApi) {
            changer(null) // 401: No one is logged in
            try {
                nouveauCaptcha()
            } catch (e: ErreurApi) {
                erreur = "Le serveur JSON est-il lancé (${Config.urlServeur}) ?"
            }
        }
    }

    suspend fun seConnecter() {
        erreur = null
        try {
            val corps = buildJsonObject {
                put("login", login.trim())
                put("password", password)
                put("captchaId", captcha?.id)
                put("captcha", reponseCaptcha)
            }
            val u = api.post("/auth/login", corps).lire<Utilisateur>()
            password = ""
            changer(u)
        } catch (e: ErreurApi) {
            erreur = e.toString() // "401 LOGIN.ERROR"
            runCatching { nouveauCaptcha() }
        }
    }

    suspend fun seDeconnecter() {
        runCatching { api.post("/auth/logout") }
        changer(null)
        runCatching { nouveauCaptcha() }
    }

    Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFECEFF1))) {
        val u = utilisateur
        if (u != null) {
            Row(Modifier.padding(8.dp), verticalAlignment = Alignment.CenterVertically) {
                Icon(Icons.Default.Person, contentDescription = null)
                Spacer(Modifier.width(8.dp))
                Text("${u.nom} (${u.role})", Modifier.weight(1f))
                TextButton(onClick = { portee.launch { seDeconnecter() } }) {
                    Icon(Icons.AutoMirrored.Filled.ExitToApp, contentDescription = null)
                    Text("Se déconnecter")
                }
            }
        } else {
            Column(Modifier.padding(8.dp)) {
                Row {
                    OutlinedTextField(login, { login = it }, label = { Text("login") }, singleLine = true, modifier = Modifier.weight(1f))
                    Spacer(Modifier.width(8.dp))
                    OutlinedTextField(
                        password,
                        { password = it },
                        label = { Text("mot de passe") },
                        singleLine = true,
                        visualTransformation = PasswordVisualTransformation(),
                        modifier = Modifier.weight(1f),
                    )
                }
                Row(verticalAlignment = Alignment.CenterVertically) {
                     // the SVG CAPTCHA image, generated by the NOTRE server (see example 22)
                    captcha?.let { ImageSvg(it.svg, hauteur = 40.dp) }
                    Spacer(Modifier.width(8.dp))
                    OutlinedTextField(reponseCaptcha, { reponseCaptcha = it }, label = { Text("captcha") }, singleLine = true, modifier = Modifier.weight(1f))
                    IconButton(onClick = { portee.launch { seConnecter() } }) {
                        Icon(Icons.Default.Check, contentDescription = "Se connecter")
                    }
                }
                erreur?.let { Text(it, color = Color.Red) }
            }
        }
    }
}

class Dictionnaire(context: Context, fichier: String) {
     // The contents of the file JSON, read once and for all
    private val racine: JsonObject = context.assets.open(fichier).bufferedReader().use { lecteur ->
        Json.parseToJsonElement(lecteur.readText()) as JsonObject
    }

     // "FORM.AJOUTE" -> rootQZXW2HTMLBWyJGT1JNIl0ZQXQZXW2HTMLBWyJBSk9VVEUiXQZQX; null if absent
    private fun chercher(cle: String): String? {
        var noeud: JsonElement? = racine
        for (partie in cle.split('.')) {
            noeud = (noeud as? JsonObject)?.get(partie)
        }
        return (noeud as? JsonPrimitive)?.takeIf { it.isString }?.content
    }

    fun t(cle: String, params: Map<String, String> = emptyMap()): String {
        val texte = chercher(cle) ?: return cle
         // {{name}} -> the value of the "name" parameter (or {{name}} if not provided)
        return Regex("\\{\\{(\\w+)\\}\\}").replace(texte) { m -> params[m.groupValues[1]] ?: m.value }
    }
}

// the dictionary for the current language: assets/i18n/<prefix>_<language>.json
@Composable
fun rememberDictionnaire(prefixe: String): Dictionnaire {
    val context = LocalContext.current
    val langue = LocalConfiguration.current.locales[0].language.let { if (it == "en") "en" else "fr" }
     // re-read only if the language changes
    return remember(langue) { Dictionnaire(context, "i18n/${prefixe}_$langue.json") }
}

class ClientsModele(private val api: Api) : ViewModel() {
    var utilisateur by mutableStateOf<Utilisateur?>(null)
        private set
    var clients by mutableStateOf(listOf<Personne>())
        private set

     // the form: idEdite == null -> create, otherwise update
    var idEdite by mutableStateOf<Int?>(null)
        private set
    private var version = 0

     // the entered values, by field name (the names expected by the server)
    var saisie by mutableStateOf(vide)
        private set

     // the error for each field (a CLÉ translation), returned by the server
    var erreurs by mutableStateOf(mapOf<String, String>())
        private set

     // the general message for the last error, and the success message
    var erreurPage by mutableStateOf<ErreurApi?>(null)
        private set
    var succes by mutableStateOf<MessageCle?>(null)
        private set

     // the request sent to the server: a Command (e.g., 23), indicating whether it is in progress
    val envoi = Commande()

    val estAdmin get() = utilisateur?.role == Role.ADMIN

    suspend fun surConnexion(u: Utilisateur?) {
        utilisateur = u
        clients = emptyList()
        if (u?.role == Role.ADMIN) recharger()
    }

    private suspend fun recharger() {
        clients = try {
            api.get("/clients").lire()
        } catch (e: ErreurApi) {
            emptyList()
        }
    }

    private fun effacerMessages() {
        erreurs = emptyMap()
        erreurPage = null
        succes = null
    }

     // a modified field: its value changes, its error disappears
    fun modifier(champ: String, valeur: String) {
        saisie = saisie + (champ to valeur)
        erreurs = erreurs - champ
    }

    fun nouveau() {
        idEdite = null
        version = 0
        saisie = vide
        effacerMessages()
    }

    fun editer(c: Personne) {
        idEdite = c.id
         // the VERSION read from the record is sent back with the modification: if
         // someone has modified the record in the meantime, the server will respond with a 409
         // CONCURRENT_UPDATE
        version = c.version
        saisie = mapOf("titre" to c.titre, "prenom" to c.prenom, "nom" to c.nom, "login" to (c.login ?: ""), "password" to "")
        effacerMessages()
    }

    suspend fun enregistrer() {
        effacerMessages()
        val corps = buildJsonObject {
            for ((champ, valeur) in saisie) put(champ, valeur)
            put("version", version)
        }
        try {
            val id = idEdite
            val client: Personne = (if (id == null) api.post("/clients", corps) else api.put("/clients/$id", corps)).lire()
            val cle = if (id == null) "FORM.AJOUTE" else "FORM.MODIFIE"
            nouveau() // form emptied, messages deleted...
             // ... except this one
            succes = MessageCle(cle, mapOf("nom" to "${client.prenom} ${client.nom}", "login" to (client.login ?: "")))
            recharger()
        } catch (e: ErreurApi) {
            erreurPage = e // the general message
            erreurs = e.champs // the error for each field
        }
    }

    suspend fun supprimer(c: Personne) {
        effacerMessages()
        try {
            api.delete("/clients/${c.id}")
            succes = MessageCle("FORM.SUPPRIME", mapOf("nom" to "${c.prenom} ${c.nom}"))
        } catch (e: ErreurApi) {
            erreurPage = e
        }
        recharger()
    }

    companion object {
        val vide = mapOf("titre" to "Mr", "prenom" to "", "nom" to "", "login" to "", "password" to "")
    }
}
