     // OBSERVÉES lists via Compose: adding, removing, or replacing an
     // element triggers recomposition. (remember, not rememberSaveable:
     // they are lost if the activity is recreated; see example 20)
    val taches = remember {
        mutableStateListOf(Tache(1, "Installer Android Studio", faite = true), Tache(2, "Lire le chapitre sur les événements"))
    }
    val journal = remember { mutableStateListOf<String>() }
    var saisie by remember { mutableStateOf("") }
    var prochainId by remember { mutableIntStateOf(3) }
     // the FocusRequester: this gives focus to the field (the cursor and the
     // phone’s keyboard) from the code

    fun tracer(message: String) {
        val heure = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm"))
        journal.add(0, "$heure $message")
    }

     // called by the ET button via the "OK" key on the keyboard (keyboardActions)
    fun ajouter() {
        val libelle = saisie.trim()
        if (libelle.isEmpty()) {
            return
        }
        taches.add(Tache(prochainId++, libelle))
        tracer("ajout de « $libelle »")
        saisie = ""
         // the focus returns to the field: you can type the next task
        focus.requestFocus()
    }

    fun basculer(tache: Tache, faite: Boolean) {
         // Save the task as a modified copy: REMPLACE
        taches[taches.indexOf(tache)] = tache.copy(faite = faite)
        tracer("« ${tache.libelle} » : ${if (faite) "faite" else "à faire"}")
    }

    fun supprimer(tache: Tache, comment: String) {
        taches.remove(tache)
        tracer("suppression de « ${tache.libelle} » ($comment)")
    }

     // Screen density: to convert pixels to dp
    val densite = LocalDensity.current.density

    PageExemple("Exemple 04 - les événements") {

                value = saisie,
                onValueChange = { saisie = it },

                singleLine = true,

                    .weight(1f)
                    .focusRequester(focus)

// possible actions. "sealed": there are no others besides those
// declared here; the compiler knows this
sealed interface ActionProfil {
    data class Modifier(val champ: Champ, val valeur: Any) : ActionProfil
    data class Quitter(val champ: Champ) : ActionProfil
    data object Majuscules : ActionProfil // a single object: the action has no data
    data object Reinitialiser : ActionProfil
}

// the reducer: it never MODIFIE uses the received state; it returns a new one.
// In each branch of the `when` statement, `action` has the type of the branch (“smart cast”):
// action.champ is accessible for a Modify
fun reducteur(etat: Etat, action: ActionProfil): Etat = when (action) {
    is ActionProfil.Modifier -> etat.copy(valeurs = etat.valeurs.avec(action.champ, action.valeur))
    is ActionProfil.Quitter -> etat.copy(touches = etat.touches + action.champ)
    ActionProfil.Majuscules -> etat.copy(
        valeurs = etat.valeurs.copy(prenom = etat.valeurs.prenom.uppercase(), nom = etat.valeurs.nom.uppercase()),
    )
    ActionProfil.Reinitialiser -> Etat(depart, emptySet())
}

import androidx.compose.runtime.getValue

// typealias: a more descriptive name for a function type
typealias Regle = (String) -> String?

// FABRIQUES rules: each returns a validation function
fun obligatoire(message: String): Regle = { v -> if (v.isBlank()) message else null }

// a rule that calls another: "?:" (the Elvis operator) returns the
// the result of the other rule, if it is not null; otherwise, the result of the next one
fun maximum(max: Int, autre: Regle): Regle = { v -> autre(v) ?: if (v.length > max) "$max caractères au plus." else null }

class ChampFormulaire(valeurInitiale: String = "", private val regle: Regle = { null }) {
     // the field’s status: observed by Compose, editable only by ICI (private set)
    var valeur by mutableStateOf(valeurInitiale)
        private set
    var erreur by mutableStateOf<String?>(null)
        private set

     // Has the user already edited the field?
    private var modifie = false

     // A keystroke: the field is validated as soon as the user modifies it
    fun changer(nouvelle: String) {
        valeur = nouvelle
        modifie = true
        erreur = regle(valeur)
    }

    fun valider(): Boolean {
        erreur = regle(valeur)
        return erreur == null
    }

     // Validate again, only if the user has already modified the field
    fun revalider() {
        if (modifie) valider()
    }
}

// vararg: any number of fields, Form(first_name, last_name, login...)
class Formulaire(private vararg val champs: ChampFormulaire) {
     // map first (TOUS the fields are validated and display their errors),
     // then all (are all fields valid?)
    fun valider(): Boolean = champs.map { it.valider() }.all { it }
}
