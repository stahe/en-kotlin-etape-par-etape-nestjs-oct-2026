// -------------------------------------------------------------------------
// A screen of an Android application is an ACTIVITY: a class that
// inherits from ComponentActivity, which the system creates when the
// application is launched. With Jetpack Compose, the activity does almost nothing:
// it hands its screen over to @Composable functions (setContent).
// A @Composable function DESCRIBES a piece of the screen (a text, a
// button, a column...) by calling other @Composable functions:
// they form a TREE. Compose executes these functions to build
// the screen; it EXECUTES THEM AGAIN when a piece of data they read (a "state")
// changes: this is RECOMPOSITION. We never modify the screen ourselves.
// =========================================================================

@Composable
fun CalculatorPage() {
    // the state: what the user has typed or chosen
    var label by rememberSaveable { mutableStateOf("Pen") }
    var priceText by rememberSaveable { mutableStateOf("2.5") } // the text of the field, as typed
    var quantity by rememberSaveable { mutableIntStateOf(1) }
    var vatRate by rememberSaveable { mutableDoubleStateOf(20.0) }
    // the history of the last 5 changes of the total incl. VAT: an UNMODIFIABLE
    // list, replaced by a new list on each change
    var history by rememberSaveable { mutableStateOf(listOf<String>()) }

    // COMPUTED values: simple local variables, recomputed on
    // each execution of the function
    val priceExclVat = price(priceText)
    val totalExclVat = quantity * priceExclVat
    val vatAmount = totalExclVat * vatRate / 100
    val inclVat = totalExclVat + vatAmount

    // ALL the changes of the state go through this local function: it
    // notes the total incl. VAT before the change, applies the change, and
    // completes the history if the total has changed
    fun change(update: () -> Unit) {
        val old = inclVat
        update()
        // the state variables ALREADY have their new value
        val newTotal = totalInclVat(price(priceText), quantity, vatRate)
        if (newTotal != old) {
            history = (listOf("${euros.format(old)} -> ${euros.format(newTotal)}") + history).take(5)
        }
    }

    // the title of the application FOLLOWS the state, like everything else
    TaskTitle("Incl. VAT: ${euros.format(inclVat)}")
    val totalStyle = MaterialTheme.typography.titleMedium.copy(fontWeight = FontWeight.Bold)

    ExamplePage("Example 02 - state") {
        Card {
            Column(Modifier.padding(16.dp)) {
                // a CONTROLLED field: it displays value, and calls onValueChange
                // on each keystroke, with the new text
                OutlinedTextField(
                    value = label,
                    // changing the item resets the quantity to 1: it is done where
                    // we know WHY the state changes
                    onValueChange = { newValue ->
                        change {
                            label = newValue
                            quantity = 1
                        }
                    },
                    label = { Text("Item") },
                    modifier = Modifier.fillMaxWidth(),
                )
                OutlinedTextField(
                    value = priceText,
                    onValueChange = { text -> change { priceText = text } },
                    label = { Text("Unit price excl. VAT") },
                    // the phone's numeric keyboard
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.Decimal),
                    modifier = Modifier.fillMaxWidth(),
                )
                Spacer(Modifier.height(12.dp))
                Row(verticalAlignment = Alignment.CenterVertically) {
                    // weight(1f): the text takes all the remaining width
                    Text("Quantity (back to 1 if the item changes)", Modifier.weight(1f))
                    // enabled = false: the button is disabled
                    IconButton(onClick = { change { quantity-- } }, enabled = quantity > 0) {
                        Icon(Icons.Default.KeyboardArrowDown, contentDescription = "minus")
                    }
                    Text("$quantity", style = totalStyle)
                    IconButton(onClick = { change { quantity++ } }) {
                        Icon(Icons.Default.Add, contentDescription = "plus")
                    }
                }
                Spacer(Modifier.height(8.dp))
                // buttons of which only ONE is selected: the VAT rate
                SingleChoiceSegmentedButtonRow(Modifier.fillMaxWidth()) {
                    rates.forEachIndexed { i, t ->
                        SegmentedButton(
                            selected = t == vatRate,
                            onClick = { change { vatRate = t } },
                            shape = SegmentedButtonDefaults.itemShape(index = i, count = rates.size),
                        ) {
                            Text("${t.toString().removeSuffix(".0")} %")
                        }
                    }
                }
                Spacer(Modifier.height(12.dp))
                OutlinedButton(onClick = {
                    change {
                        label = "Pen"
                        priceText = "2.5"
                        quantity = 1
                        vatRate = 20.0
                    }
                }) {
                    Text("Reset")
                }
            }
        }
        Card {
            Column(Modifier.padding(16.dp)) {
                Text("$quantity x $label", style = MaterialTheme.typography.titleLarge)
                HorizontalDivider(Modifier.padding(vertical = 8.dp))
                Line("Total excl. VAT", euros.format(totalExclVat))
                Line("VAT ($vatRate %)", euros.format(vatAmount))
                Line("Total incl. VAT", euros.format(inclVat), totalStyle)
            }

                // one line of text per element of the history (cf. example 03)
                for (line in history) {
                    Text("• $line")
                }
            }
        }

@Composable
fun TasksPage() {
    // lists OBSERVED by Compose: adding, removing or replacing an
    // element causes the recomposition. (remember, and not rememberSaveable:
    // they are lost if the activity is recreated, cf. example 20)
    val tasks = remember {
        mutableStateListOf(Task(1, "Install Android Studio", done = true), Task(2, "Read the chapter on events"))
    }
    val log = remember { mutableStateListOf<String>() }
    var input by remember { mutableStateOf("") }
    var nextId by remember { mutableIntStateOf(3) }
    // the FocusRequester: a way to give the focus to the field (the cursor and the
    // phone's keyboard) from the code
    val focus = remember { FocusRequester() }

    fun trace(message: String) {
        val time = LocalTime.now().format(DateTimeFormatter.ofPattern("HH:mm"))
        log.add(0, "$time $message")
    }

    // called by the button AND by the keyboard's "OK" key (keyboardActions)
    fun add() {
        val label = input.trim()
        if (label.isEmpty()) {
            return
        }
        tasks.add(Task(nextId++, label))
        trace("added \"$label\"")
        input = ""
        // the focus returns to the field: the next task can be typed
        focus.requestFocus()
    }

    fun toggle(task: Task, done: Boolean) {
        // we REPLACE the task with a modified copy
        tasks[tasks.indexOf(task)] = task.copy(done = done)
        trace("\"${task.label}\": ${if (done) "done" else "to do"}")
    }

    fun remove(task: Task, how: String) {
        tasks.remove(task)
        trace("removed \"${task.label}\" ($how)")
    }

    // the screen density: to convert pixels into dp
    val density = LocalDensity.current.density

    ExamplePage("Example 04 - events") {
        Row(verticalAlignment = Alignment.CenterVertically) {
            OutlinedTextField(
                value = input,
                onValueChange = { input = it },
                placeholder = { Text("new task (Escape to clear)") },
                singleLine = true,
                // the "OK" key of the phone's keyboard adds the task
                keyboardOptions = KeyboardOptions(imeAction = ImeAction.Done),
                keyboardActions = KeyboardActions(onDone = { add() }),
                modifier = Modifier
                    .weight(1f)
                    .focusRequester(focus)
                    // a physical keyboard (emulator, tablet): Escape clears the field
                    .onPreviewKeyEvent { event ->
                        if (event.key == Key.Escape) {
                            input = ""
                            true // the event is handled
                        } else {
                            false // the event continues on its way
                        }
                    },
            )
            Spacer(Modifier.width(8.dp))
            Button(onClick = { add() }) { Text("Add") }
        }
        Spacer(Modifier.height(12.dp))
        for (t in tasks) {
            // key(t.id): the IDENTITY of the line. The state of each line (its
            // swipe) follows ITS task, even if the list changes (cf. the course)
            key(t.id) {
                TaskRow(
                    task = t,
                    onToggle = { done -> toggle(t, done) },
                    onRemove = { how -> remove(t, how) },

            }
        }
        Spacer(Modifier.height(12.dp))
        // pointerInput: any component becomes sensitive to gestures
        Box(
            contentAlignment = Alignment.Center,

            Text(line, fontSize = 12.sp)
        }
    }

// the name of each field of the form
enum class Field { TITLE, FIRSTNAME, LASTNAME, AGE, NEWSLETTER, EMAIL, COUNTRY, PRESENTATION }

// the data of the form: an UNMODIFIABLE object (everything is val). To
// "modify" it, we make a copy of it: profile.copy(lastName = "DUPONT"). The
// copy function is written by Kotlin for any "data" class
data class Profile(
    val title: String,
    val firstName: String,
    val lastName: String,
    val age: Int,
    val newsletter: Boolean,
    val email: String,
    val country: String,
    val presentation: String,
) {
    // the value of a field, designated by its name
    fun value(f: Field): Any = when (f) {
        Field.TITLE -> title
        Field.FIRSTNAME -> firstName
        Field.LASTNAME -> lastName
        Field.AGE -> age
        Field.NEWSLETTER -> newsletter
        Field.EMAIL -> email
        Field.COUNTRY -> country
        Field.PRESENTATION -> presentation
    }

    // a copy in which ONE field, designated by its name, has a new value
    fun withValue(f: Field, v: Any): Profile = when (f) {
        Field.TITLE -> copy(title = v as String)
        Field.FIRSTNAME -> copy(firstName = v as String)
        Field.LASTNAME -> copy(lastName = v as String)
        Field.AGE -> copy(age = v as Int)
        Field.NEWSLETTER -> copy(newsletter = v as Boolean)
        Field.EMAIL -> copy(email = v as String)
        Field.COUNTRY -> copy(country = v as String)
        Field.PRESENTATION -> copy(presentation = v as String)
    }

    // the JSON object of the profile: one JSON field per field of the form
    fun toJson(): JsonObject = JsonObject(
        Field.entries.associate { f ->
            val v = value(f)
            f.name.lowercase() to when (v) {
                is Int -> JsonPrimitive(v) // "is": a type test; v is then an Int
                is Boolean -> JsonPrimitive(v)
                else -> JsonPrimitive(v.toString())
            }
        },
    )
}

    title = "Mme",

// the complete state: the values, and the fields already left
data class State(val values: Profile, val touched: Set<Field>)

// the possible actions. "sealed": there are no others than those
// declared here, the compiler knows it
sealed interface ProfileAction {
    data class Update(val field: Field, val value: Any) : ProfileAction
    data class Leave(val field: Field) : ProfileAction
    data object Uppercase : ProfileAction // a unique object: the action has no data
    data object Reset : ProfileAction
}

// the reducer: it NEVER modifies the received state, it returns a new one.
// In each branch of the when, action has the type of the branch ("smart cast"):
// action.field is accessible for an Update
fun reducer(state: State, action: ProfileAction): State = when (action) {
    is ProfileAction.Update -> state.copy(values = state.values.withValue(action.field, action.value))
    is ProfileAction.Leave -> state.copy(touched = state.touched + action.field)
    ProfileAction.Uppercase -> state.copy(
        values = state.values.copy(firstName = state.values.firstName.uppercase(), lastName = state.values.lastName.uppercase()),
    )
    ProfileAction.Reset -> State(initial, emptySet())
}

// the account, as typed
data class Account(
    val title: String,
    val firstName: String,
    val lastName: String,
    val login: String,
    val password: String,
    val confirmation: String,
)

val titles = listOf("Mr", "Mme", "Melle")
val loginFormat = Regex("^[a-z0-9._-]{1,30}$")
const val PASSWORD_MIN = 6

// the rules: a pure function, which only depends on its argument. It
// returns, for each field in error, its message. buildMap: we fill
// a modifiable Map, the function returns an UNMODIFIABLE Map
fun validate(a: Account): Map<Field, String> = buildMap {
    if (a.title !in titles) put(Field.TITLE, "Choose a title from the list.")
    when {
        a.firstName.isBlank() -> put(Field.FIRSTNAME, "The first name is required.")
        a.firstName.length > 20 -> put(Field.FIRSTNAME, "The first name must not exceed 20 characters.")
    }
    when {
        a.lastName.isBlank() -> put(Field.LASTNAME, "The last name is required.")
        a.lastName.length > 20 -> put(Field.LASTNAME, "The last name must not exceed 20 characters.")
    }
    when {
        a.login.isEmpty() -> put(Field.LOGIN, "The login is required.")
        !loginFormat.matches(a.login) -> put(Field.LOGIN, "Unaccented lower-case letters, digits, dot, hyphen, underscore.")
    }
    when {
        a.password.isEmpty() -> put(Field.PASSWORD, "The password is required.")
        a.password.length < PASSWORD_MIN -> put(Field.PASSWORD, "At least $PASSWORD_MIN characters.")
    }
    // a rule that compares TWO fields
    if (a.confirmation != a.password) put(Field.CONFIRMATION, "The confirmation does not match the password.")
}

import androidx.compose.runtime.getValue
import androidx.compose.runtime.mutableStateOf
import androidx.compose.runtime.setValue

// typealias: a more meaningful name for a function type
typealias Rule = (String) -> String?

// rule FACTORIES: each one returns a validation function
fun required(message: String): Rule = { v -> if (v.isBlank()) message else null }

// a rule that COMPLETES another one: "?:" (the Elvis operator) gives the
// result of other, if it is not null, otherwise that of the rest
fun maximum(max: Int, other: Rule): Rule = { v -> other(v) ?: if (v.length > max) "$max characters at most." else null }

class FormField(initialValue: String = "", private val rule: Rule = { null }) {
    // the state of the field: observed by Compose, modifiable only HERE (private set)
    var value by mutableStateOf(initialValue)
        private set
    var error by mutableStateOf<String?>(null)
        private set

    // has the user already modified the field?
    private var modified = false

    // a keystroke: the field validates itself as soon as the user has modified it
    fun change(newValue: String) {
        value = newValue
        modified = true
        error = rule(value)
    }

    fun validate(): Boolean {
        error = rule(value)
        return error == null
    }

    // revalidate, only if the user has already modified the field
    fun revalidate() {
        if (modified) validate()
    }
}

// vararg: any number of fields, Form(firstName, lastName, login...)
class Form(private vararg val fields: FormField) {
    // map first (ALL the fields are validated, and display their error),
    // all next (are they all fine?)
    fun validate(): Boolean = fields.map { it.validate() }.all { it }
}

// the fields of the form, and their rules
class AccountForm {
    var title by mutableStateOf("Mr")
    val firstName = FormField(rule = maximum(20, required("The first name is required.")))
    val lastName = FormField(rule = maximum(20, required("The last name is required.")))

    // a rule written directly: a lambda
    val login = FormField(
        rule = { v ->
            when {
                v.isEmpty() -> "The login is required."
                !Regex("^[a-z0-9._-]{1,30}$").matches(v) -> "Unaccented lower-case letters, digits, dot, hyphen, underscore."
                else -> null
            }
        },
    )
    val password = FormField(
        rule = { v ->
            when {
                v.isEmpty() -> "The password is required."
                v.length < 6 -> "At least 6 characters."
                else -> null
            }
        },
    )

    // a rule that reads ANOTHER field
    val confirmation = FormField(
        rule = { v -> if (v == password.value) null else "The confirmation does not match the password." },
    )
    val form = Form(firstName, lastName, login, password, confirmation)

    // the values, in JSON (what would be sent to the server)
    fun toJson(): String = JsonObject(
        mapOf(
            "title" to JsonPrimitive(title),
            "firstName" to JsonPrimitive(firstName.value.trim()),
            "lastName" to JsonPrimitive(lastName.value.trim()),
            "login" to JsonPrimitive(login.value),
            "password" to JsonPrimitive(password.value),
        ),
    ).toString()
}

package fr.stahe.examples.ex09

// a time slot, as described by the RdvMedecins API
data class Slot(val startHour: Int, val startMinute: Int, val endHour: Int, val endMinute: Int)

enum class Part { ALL, START, END }

// "8:05" rather than "8:5"; private: invisible outside this file
private fun time(h: Int, m: Int) = "$h:${m.toString().padStart(2, '0')}"

// an OPTIONAL parameter, with its default value: part = Part.ALL
fun timeRange(s: Slot, part: Part = Part.ALL): String = when (part) {
    Part.START -> time(s.startHour, s.startMinute)
    Part.END -> time(s.endHour, s.endMinute)
    Part.ALL -> "${time(s.startHour, s.startMinute)} - ${time(s.endHour, s.endMinute)}"
}

// EXTENSIONS of the String type: new properties for ALL the
// strings. In the body, this designates the string
// "marie pelissier" -> "Marie Pelissier"; \p{L}: a letter, accented or not
val String.capitalized: String
    get() = Regex("\\p{L}+").replace(this) { m ->
        m.value.lowercase().replaceFirstChar { it.uppercase() }
    }

// "Marie PELISSIER" -> "MP"
val String.initials: String
    get() = split(Regex("\\s+")).filter { it.isNotEmpty() }.joinToString("") { it.first().uppercase() }
