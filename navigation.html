    5. Navigation

    An application has multiple screens. In [Android], along with [Compose], an activity is not opened per screen: a SEULE activity displays the application’s screens in turn, controlled by a ROUTEUR. This is the [navigation-compose] library for [Jetpack], the equivalent of the [go_router] package for [Flutter].

    5.1. Example 17: The Router

// ex17/Routes.kt - the ROUTES of the application
// -------------------------------------------------------------------------
// An application has multiple screens ("pages"). The ROUTEUR associates a
// ROUTE to each screen. The Jetpack Navigation library (navigation-
// composes) manages a PILE of screens: a screen is pushed onto the stack (navigate), and the
// phone’s Back button pops it off the stack (popBackStack).
// A route is a Kotlin object, @Serializable:
//    - an “object” for a screen with no parameters (Home);
//    - a "data class" for a screen with parameters: DetailMedecin(id = 2).
// The type of the parameters is checked at compile time: DetailMedecin("2")
// would be rejected. (Navigation knows how to stack these objects, hence
// @Serializable.)
// Each screen can also have a ADRESSE, a “deep link”
// of the type app://examples/doctors/2: this is the closest equivalent,
// on Android, to web addresses (see Ex17Activity.kt).
// =========================================================================
package fr.stahe.exemples.ex17

import kotlinx.serialization.Serializable

@Serializable
object Accueil

@Serializable
object ListeMedecins

// a route parameter
@Serializable
data class DetailMedecin(val id: Int)

// a parameter FACULTATIF (with a default value): ?q=... in the URL
@Serializable
data class Recherche(val q: String = "")

@Serializable
object APropos

// an old URL, which REDIRIGE links to the list of doctors
@Serializable
object Docteurs

// the unknown addresses screen
@Serializable
data class PageIntrouvable(val adresse: String)

// the beginning of all addresses in this example
const val BASE = "app://exemples"

// ex17/Ex17Activity.kt - NavHost: the routes and their screens
// -------------------------------------------------------------------------
//    - rememberNavController(): the navigation controller (the stack);
//    - NavHost(nav, startDestination = Home) { ... }: the
//      current route. The block describes the GRAPHE navigation: a
//      composable <Route> entry { ... } per route;
//    - entree.toRoute<DetailMedecin>() : the screen’s route object, with
//      its parameters;
//    - deepLinks: the screen’s address. Thanks to the intent filter of
//      (AndroidManifest.xml), an address such as app://examples/...
//      can even be opened from outside the app:
//          adb shell am start -a android.intent.action.VIEW -d "app://examples/doctors/2"

class Ex17Activity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        enableEdgeToEdge()
        setContent { ThemeExemples { Application() } }
    }
}

@Composable
fun Application() {
    val nav = rememberNavController()
    Coquille(nav) {
        NavHost(navController = nav, startDestination = Accueil) {
            composable<Accueil>(deepLinks = listOf(navDeepLink<Accueil>(basePath = "$BASE/accueil"))) {
                PageAccueil(onLien = { nav.ouvrirLien(it) }, onRedirection = { nav.navigate(Docteurs) })
            }
            composable<ListeMedecins>(deepLinks = listOf(navDeepLink<ListeMedecins>(basePath = "$BASE/medecins"))) {
                 // navigate: the doctor screen is EMPILÉ above the list
                PageListeMedecins(onVoir = { id -> nav.navigate(DetailMedecin(id)) })
            }
             // app://examples/doctors/2: the id parameter is taken from the address
            composable<DetailMedecin>(deepLinks = listOf(navDeepLink<DetailMedecin>(basePath = "$BASE/medecins"))) { entree ->
                val route = entree.toRoute<DetailMedecin>()
                PageDetailMedecin(
                    id = route.id,
                     // popBackStack: navigate back through the screen (like the Back button); if
                     // there is nothing below it (screen opened via a URL), go to the list
                    onRetour = { if (!nav.popBackStack()) nav.navigate(ListeMedecins) },
                     // REMPLACER the current screen (no stacking): unstack the
                     // current doctor (inclusive) before stacking the next one
                    onSuivant = { suivant ->
                        nav.navigate(DetailMedecin(suivant)) { popUpTo<DetailMedecin> { inclusive = true } }
                    },
                )
            }
             // app://examples/search?q=pediatrician
            composable<Recherche>(deepLinks = listOf(navDeepLink<Recherche>(basePath = "$BASE/recherche"))) { entree ->
                PageRecherche(q = entree.toRoute<Recherche>().q, onRechercher = { saisie -> nav.navigate(Recherche(saisie)) })
            }
            composable<APropos> { PageAPropos() }
             // a REDIRECTION: the screen displays nothing; it immediately replaces
             // its own route with another
            composable<Docteurs> {
                LaunchedEffect(Unit) {
                    nav.navigate(ListeMedecins) { popUpTo<Docteurs> { inclusive = true } }
                }
            }
            composable<PageIntrouvable> { entree ->
                EcranIntrouvable(adresse = entree.toRoute<PageIntrouvable>().adresse, onAccueil = { nav.allerA(Accueil) })
            }
        }
    }
}

    • lines 46–48: [rememberNavController()]: the navigation controller (the stack); [NavHost(nav, startDestination = Accueil)] displays the screen for the current route;

    • lines 66–67: REMPLACER is the current screen: the current doctor ([popUpTo<DetailMedecin> [ inclusive = true ]]) is popped before the next one is pushed;

// a tab: the route it leads to, its label, its icon, and the test that
// indicates whether it is active for a given screen
class Onglet(val route: Any, val libelle: String, val icone: ImageVector, val actif: (NavDestination) -> Boolean)

val onglets = listOf(
    Onglet(Accueil, "Accueil", Icons.Default.Home) { it.hasRoute<Accueil>() },
     // the "Doctors" tab remains active on a doctor’s screen
    Onglet(ListeMedecins, "Médecins", Icons.Default.Person) { it.hasRoute<ListeMedecins>() || it.hasRoute<DetailMedecin>() },
    Onglet(Recherche("généraliste"), "Recherche", Icons.Default.Search) { it.hasRoute<Recherche>() },
    Onglet(APropos, "À propos", Icons.Default.Info) { it.hasRoute<APropos>() },
)

// the address of the current screen, to display it
fun adresseDe(entree: NavBackStackEntry?): String {
    val d = entree?.destination ?: return ""
    return when {
        d.hasRoute<Accueil>() -> "/"
        d.hasRoute<ListeMedecins>() -> "/medecins"
        d.hasRoute<DetailMedecin>() -> "/medecins/${entree.toRoute<DetailMedecin>().id}"
        d.hasRoute<Recherche>() -> "/recherche?q=${entree.toRoute<Recherche>().q}"
        d.hasRoute<APropos>() -> "/a-propos"
        d.hasRoute<PageIntrouvable>() -> entree.toRoute<PageIntrouvable>().adresse
        else -> "?"
    }
}

// Go to a tab: the stack is REMPLACÉE, starting with that tab
fun NavHostController.allerA(route: Any) {
    navigate(route) {
         // Unstack everything, down to the first screen (excluded)...
        popUpTo(graph.findStartDestination().id)
         // ... and the same screen is not stacked twice
        launchSingleTop = true
    }
}

// follow a LIEN (an address); an unknown address leads to PageIntrouvable
fun NavHostController.ouvrirLien(adresse: String) {
    try {
        navigate(Uri.parse("$BASE$adresse"))
    } catch (e: IllegalArgumentException) {
        navigate(PageIntrouvable(adresse))
    }
}

@Composable
fun Coquille(nav: NavHostController, contenu: @Composable () -> Unit) {
    val entree by nav.currentBackStackEntryAsState()
    val destination = entree?.destination
    Scaffold(
        topBar = { TopAppBar(title = { Text("Exemple 17 - le routeur") }) },
         // NavigationBar: the bottom tab bar, typical of a mobile device
        bottomBar = {
            NavigationBar {
                for (o in onglets) {
                    NavigationBarItem(
                        selected = destination != null && o.actif (destination),
                        onClick = { nav.allerA(o.route) },
                        icon = { Icon(o.icone, contentDescription = null) },
                        label = { Text(o.libelle) },
                    )
                }
            }
        },
    ) { marges ->
        Column(Modifier.padding(marges).fillMaxSize()) {
            Text(
                "adresse courante : ${adresseDe(entree)}",
                fontFamily = FontFamily.Monospace,
                fontSize = 12.sp,
                modifier = Modifier.fillMaxWidth().background(Color(0xFFF5F5F5)).padding(8.dp),
            )
            Box(Modifier.padding(16.dp).weight(1f)) { contenu() }
        }
    }
}

// a page with a parameter: the router passes it an `id`, an `Int`
@Composable
fun PageDetailMedecin(id: Int, onRetour: () -> Unit, onSuivant: (Int) -> Unit) {
     // find: the first suitable doctor, or null
    val medecin = medecins.find { it.id == id }
    if (medecin == null) {
        Text("Le médecin n° $id n'existe pas.", color = Color.Red)
        return
    }
    val suivant = medecin.id % medecins.size + 1
    TitreTache("${medecin.prenom} ${medecin.nom}")
    Column {
        Text("${medecin.titre} ${medecin.prenom} ${medecin.nom}", style = MaterialTheme.typography.titleLarge)
        Text("Spécialité : ${medecin.specialite}")
        Spacer(Modifier.height(16.dp))
        FlowRow(horizontalArrangement = Arrangement.spacedBy(8.dp)) {
            OutlinedButton(onClick = onRetour) { Text("Retour") }
            Button(onClick = { onSuivant(suivant) }) { Text("Médecin suivant") }
        }
    }
}

// a page driven by its route: Search(q = "pediatrician"). Search,
// this is ALLER for the Search(q = <input>) route: the page displays what
// the route specifies, and nothing else (the Back button returns to the
// previous search).
// =========================================================================
@Composable
fun PageRecherche(q: String, onRechercher: (String) -> Unit) {
    TitreTache("Recherche")
    val resultats = medecins.filter { "${it.prenom} ${it.nom} ${it.specialite}".lowercase().contains(q.lowercase()) }
     // The input starts with the route’s value; rememberSaveable(q):
     // if the route changes (Back button), the entry restarts with the new value
    var saisie by rememberSaveable(q) { mutableStateOf(q) }
    Column {
        OutlinedTextField(
            value = saisie,
            onValueChange = { saisie = it },
            placeholder = { Text("nom ou spécialité") },
            trailingIcon = { Icon(Icons.Default.Search, contentDescription = null) },
            singleLine = true,
             // the “Search” key on the keyboard
            keyboardOptions = KeyboardOptions(imeAction = ImeAction.Search),
            keyboardActions = KeyboardActions(onSearch = { onRechercher(saisie.trim()) }),
            modifier = Modifier.fillMaxWidth(),
        )
        Spacer(Modifier.height(8.dp))
        for (m in resultats) {
            ListItem(headlineContent = { Text("${m.prenom} ${m.nom} (${m.specialite})") })
        }
        if (resultats.isEmpty()) {
            ListItem(headlineContent = { Text("Aucun résultat.", color = Color.Gray) })
        }
    }
}

     // status: null, no one is logged in. private set: can only be modified here

// A GARDE is a function that receives the logged-in user (or null) and

//    - null: the screen can be displayed;

package fr.stahe.exemples.ex18

@Serializable
data class Accueil(val refus: String? = null)

@Serializable

const val BASE = "app://exemples18"

typealias Garde = (Utilisateur?) -> Any?

// the "login required" guard: not logged in -> to the login page,
// retaining the address of the requested screen
fun connexionRequise(adresse: String): Garde = { u -> if (u == null) Connexion(redirect = adresse) else null }

// "role required" guard: logged in, but not with the correct role -> return to
// the home page, with a reason
fun roleRequis(adresse: String, vararg roles: Role): Garde = { u ->
    if (u == null || u.role !in roles) Home(refus = address) else null
}

// the guards, in order: the first one to return a route wins
@Composable
fun Protege(nav: NavHostController, vararg gardes: Garde, contenu: @Composable () -> Unit) {
    val redirection = gardes.firstNotNullOfOrNull { garde -> garde(Session.utilisateur) }
    if (redirection == null) {
        contenu()
    } else {
         // navigation is an “effect”: we do not navigate PENDANT the display
        LaunchedEffect(redirection) {
            val ici = nav.currentBackStackEntry?.destination?.id ?: return@LaunchedEffect
             // the rejected screen is replaced (including it) by the redirection
            nav.navigate(redirection) { popUpTo(ici) { inclusive = true } }
        }
    }
}

                Protege(nav, connexionRequise("$BASE/administration"), roleRequis("/administration", Role.ADMIN)) {
                    PageAdministration()

@Composable
fun PageConnexion(redirection: String?, onConnecte: (String?) -> Unit) {
    var login by remember { mutableStateOf("") }
    var password by remember { mutableStateOf("") }
    var erreur by remember { mutableStateOf(false) }

    fun valider() {
        if (!Session.connecter(login.trim(), password)) {
            erreur = true
            return
        }
        onConnecte(redirection)
    }

    Column {
        if (redirection != null) {
            Card(colors = CardDefaults.cardColors(containerColor = Color(0xFFFFF8E1))) {
                Text("Connectez-vous pour ouvrir ${redirection.removePrefix(BASE)}.", Modifier.padding(12.dp))
            }
        }
        OutlinedTextField(value = login, onValueChange = { login = it }, label = { Text("login") }, singleLine = true, modifier = Modifier.fillMaxWidth())
        OutlinedTextField(
            value = password,
            onValueChange = { password = it },
            label = { Text("mot de passe") },
            singleLine = true,
            visualTransformation = PasswordVisualTransformation(),
            keyboardOptions = KeyboardOptions(imeAction = ImeAction.Go),
            keyboardActions = KeyboardActions(onGo = { valider() }),
            modifier = Modifier.fillMaxWidth(),
        )
        Spacer(Modifier.height(16.dp))
        Button(onClick = { valider() }) { Text("Se connecter") }
        if (erreur) {
            Text("Login ou mot de passe incorrect.", color = Color.Red, modifier = Modifier.padding(top = 8.dp))
        }
    }
